Self-contained script that predicts electrical consumption by Area & Appliance and flags unexplained bill spikes

import pandas as pd
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error, r2_score

# Area & Appliance-wise IoT Electrical Load Dataset
data = [
    ("Bedroom",     "Split AC (1.5 Ton)",        6.8, 7.5,  5.0, 11.95, 14.80),
    ("Bathroom",    "Water Geyser (2000W)",      8.7, 2.5,  8.5,  5.25,  9.40),
    ("Kitchen",     "Double-Door Refrigerator",  1.4, 9.0, 15.0,  3.35,  3.45),
    ("Kitchen",     "Microwave Oven (1200W)",    5.2, 0.8,  2.0,  1.02,  1.05),
    ("Living Room", "Smart LED TV & Soundbar",   0.9, 5.5, 10.0,  1.38,  1.42),
    ("Living Room", "Tower Air Cooler",          2.2, 6.0,  4.0,  3.15,  3.20),
    ("Home Office", "Desktop PC & Dual Monitor", 1.8, 6.5,  9.5,  2.95,  4.65),
    ("Bedroom",     "Ceiling Fan & LED Lights",  0.5, 10.0, 2.0,  1.18,  1.20),
    ("Utility Area","Washing Machine (Front)",   4.5, 1.5,  3.0,  1.65,  1.70),
    ("Utility Area","Submersible Water Pump",    6.5, 1.2,  1.0,  1.82,  3.60),
    ("Bedroom",     "Room Heater (Oil Fin)",     8.2, 4.0,  6.0,  7.75, 10.20),
    ("Kitchen",     "Induction Cooktop",         7.5, 1.5,  2.5,  2.65,  2.70),
]

df = pd.DataFrame(data, columns=[
    "Area", "Appliance", "Current_A", "Active_Hours",
    "Standby_Hours", "Expected_kWh", "Measured_kWh"
])
df["Amp_Hours"] = df["Current_A"] * df["Active_Hours"]

X = df[["Amp_Hours", "Current_A", "Standby_Hours"]]
y = df["Expected_kWh"]

model = LinearRegression()
model.fit(X, y)
df["Predicted_kWh"] = model.predict(X).round(2)
df["Spike_Waste_kWh"] = (df["Measured_kWh"] - df["Predicted_kWh"]).round(2)

print("R2 Score:", round(r2_score(y, df["Predicted_kWh"]), 4))
print(df[["Area", "Appliance", "Current_A", "Predicted_kWh", "Measured_kWh", "Spike_Waste_kWh"]])
